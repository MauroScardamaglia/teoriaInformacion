from math import log
from mis_funciones import extender_dist_prob

TOLERANCIA_FLOAT = 1e-9

def calcular_entropia_fuente(probabilidades, base=2):
    entropia = 0
    if type(probabilidades) == dict:
        for item in probabilidades:
            p = probabilidades[item]
            if p > 0:
                entropia += p * calcular_cantidad_informacion(p, base)
    else:
        for i in range(len(probabilidades)):
            p = probabilidades[i]
            if p > 0:
                entropia += p * calcular_cantidad_informacion(p, base)
    return entropia

def calcular_cantidad_informacion(prob, base=2):
    if prob <= 0 or abs(prob - 1.0) < TOLERANCIA_FLOAT:
        return 0
    return log(1 / prob, base)

def long_media_codigo(palabras_codigo, probabilidades):
    longitud = 0.0
    longitudes = longitudes_palabras(palabras_codigo)
    for i in range(len(longitudes)):
        longitud += probabilidades[i] * longitudes[i]
    return longitud

def longitudes_palabras(palabras_codigo):
    longitudes = []
    for palabra in palabras_codigo:
        longitudes.append(len(palabra))
    return longitudes

def cumple_shannon(dist_prob, extension, orden):
    dist_prob_extension = extender_dist_prob(dist_prob, orden)
    entropia_fuente = calcular_entropia_fuente(dist_prob)
    long_media = long_media_codigo(extension, dist_prob_extension)

    return (entropia_fuente <= long_media / orden) and (
        long_media / orden < entropia_fuente + 1 / orden
    )

            p = probabilidades[i]
            if p > 0:

def long_media_codigo(palabras_codigo, probabilidades):

    longitudes = longitudes_palabras(palabras_codigo)
    for i in range(len(longitudes)):
        longitud += probabilidades[i] * longitudes[i]
    return longitud

def longitudes_palabras(palabras_codigo):
    longitudes = []
    for palabra in palabras_codigo:
        longitudes.append(len(palabra))
    return longitudes

def extender_alfabeto(alfabeto, n):
    if n <= 0:
        return []
    elif n == 1:
        return alfabeto
    else:
        alfabeto_anterior = extender_alfabeto(alfabeto, n - 1)
        alfabeto_extend = []
        for i in range(len(alfabeto_anterior)):
            for j in range(len(alfabeto)):
                alfabeto_extend.append(alfabeto_anterior[i] + alfabeto[j])
        return alfabeto_extend
